% ---------------------------------------------------------------------------
% Title page
% ---------------------------------------------------------------------------
\newcommand{\maketitlepage}{%
  % The title page needs no hyperref anchor, and suppressing it avoids a
  % duplicate page.1 destination once front matter switches to roman numerals.
  \hypersetup{pageanchor=false}%
  \begin{titlepage}
    \thispagestyle{titlestyle}
    \centering
    \vspace*{1.5cm}

    {\Huge\bfseries \reporttitle\par}
    \vspace{0.8cm}
    {\Large \reportsubtitle\par}

    \vspace{1.5cm}
    {\large \coursename\par}
    \vspace{0.3cm}
    {\large \textbf{Group \groupnumber}\par}

    \vspace{1.2cm}
    {\large\bfseries Group Members\par}
    \vspace{0.4cm}
    \groupmemberstable

    \vfill
  \end{titlepage}
  % Front matter in roman numerals so the title page and the body do not both
  % claim page 1 (which produces duplicate hyperref anchors).
  \pagenumbering{roman}%
  \hypersetup{pageanchor=true}%
}
